WITH series_counts AS (
  SELECT
    workforce_snapshot.occupational_series_code,
    workforce_snapshot.occupational_series,
    COUNT(*) AS headcount,
    SUM(CASE WHEN workforce_snapshot.length_of_service_years >= 20 THEN 1 ELSE 0 END) AS headcount_20_plus_years
  FROM workforce.semantic.workforce_snapshot
  WHERE workforce_snapshot.snapshot_month = DATE '2026-08-01'
    AND workforce_snapshot.agency_subelement_code = 'HSBD'
    AND workforce_snapshot.occupational_series_code IS NOT NULL
    AND workforce_snapshot.occupational_series IS NOT NULL
  GROUP BY
    workforce_snapshot.occupational_series_code,
    workforce_snapshot.occupational_series
)
SELECT
  series_counts.occupational_series_code,
  series_counts.occupational_series,
  series_counts.headcount,
  series_counts.headcount_20_plus_years,
  SUM(series_counts.headcount) OVER () AS total_headcount_all_series
FROM series_counts
ORDER BY series_counts.headcount DESC, series_counts.occupational_series_code, series_counts.occupational_series

WITH `filtered` AS (
  SELECT
    DATE_TRUNC('MONTH', `open_date`) AS `open_month`,
    CASE
      WHEN CAST(`minimum_grade` AS INT) BETWEEN 5 AND 9 THEN '5 to 9'
      WHEN CAST(`minimum_grade` AS INT) BETWEEN 10 AND 12 THEN '10 to 12'
      WHEN CAST(`minimum_grade` AS INT) BETWEEN 13 AND 14 THEN '13 to 14'
      WHEN CAST(`minimum_grade` AS INT) >= 15 THEN '15 and above'
      ELSE NULL
    END AS `minimum_grade_band`,
    COALESCE(`total_openings`, 0) AS `stated_openings`
  FROM `workforce`.`semantic`.`job_posting`
  WHERE `agency_subelement_code` = 'HSBD'
    AND `open_date` >= DATE '2025-09-01'
    AND `open_date` < DATE '2026-11-01'
    AND `open_date` IS NOT NULL
    AND `minimum_grade` IS NOT NULL
    AND array_contains(`occupational_series_codes`, '1895')
), `grouped` AS (
  SELECT
    `open_month`,
    `minimum_grade_band`,
    COUNT(*) AS `postings`,
    SUM(`stated_openings`) AS `total_openings`
  FROM `filtered`
  WHERE `minimum_grade_band` IS NOT NULL
  GROUP BY `open_month`, `minimum_grade_band`
)
SELECT
  `open_month`,
  `minimum_grade_band`,
  `postings`,
  `total_openings`,
  SUM(`postings`) OVER (PARTITION BY `open_month`) AS `month_total_postings`
FROM `grouped`
ORDER BY
  `open_month`,
  CASE `minimum_grade_band`
    WHEN '5 to 9' THEN 1
    WHEN '10 to 12' THEN 2
    WHEN '13 to 14' THEN 3
    WHEN '15 and above' THEN 4
    ELSE 5
  END
